#### linux连接串口
```
#先通过串口工具进入开发版的shell
sudo apt-get install minicom
ls /dev/ttyUSB*
sudo minicom -s
```

#### 连接无线网络
```
#启动无线网卡
ifconfig wlan0 up
#搜索wifi热点
iw wlan0 scan | grep SSID
#修改连接的ssid和密码
vi /etc/wpa_supplicant.conf
#连接并自动获取ip
wpa_supplicant -Dnl80211 -c /etc/wpa_supplicant.conf -i wlan0 & udhcpc -i wlan0

```
